─────────────
self.public       # সবাই access করতে পারে
self._protected   # convention, বাইরে থেকে avoid করো
self.__private    # name mangling, বাইরে থেকে access কঠিন

@property
def x(self): return self.__x      # getter

@x.setter  
def x(self, val): self.__x = val  # setter
